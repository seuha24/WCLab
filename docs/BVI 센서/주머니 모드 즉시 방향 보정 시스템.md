# SafeLight 주머니 모드 즉시 방향 보정 시스템

## 📱 문제 정의 및 해결 방안

### 1.1 핵심 문제
- **지속적 휴대 불가**: 사용자가 계속 핸드폰을 들고 다닐 수 없음
- **주머니 모드 오차**: 주머니에 넣으면 YawRate(z축) 값이 부정확
- **누적 오차**: 자이로 적분값이 시간이 지날수록 틀어짐
- **즉시성 요구**: 꺼냈을 때 바로 정확한 방향 필요

### 1.2 해결 전략
```
주머니 모드 → GPS 이동 추적 → 필요시 꺼냄 → 즉시 보정 → 정확한 방향
```

## 2. 🎯 즉시 방향 보정 알고리즘

### 2.1 GPS 기반 즉시 방향 보정 (GPS Heading Calibration)

#### 핵심 원리
```dart
// GPS 두 지점을 이용한 자기장 간섭 없는 방향 계산
class GPSHeadingCalibration {
  Position? lastPosition;
  Position? currentPosition;
  double? gpsHeading;
  
  // GPS 이동 방향 계산 (자기장 간섭 없음!)
  double calculateGPSHeading() {
    if (lastPosition != null && currentPosition != null) {
      double deltaLat = currentPosition!.latitude - lastPosition!.latitude;
      double deltaLon = currentPosition!.longitude - lastPosition!.longitude;
      
      // GPS bearing 계산
      double heading = atan2(deltaLon * cos(currentPosition!.latitude),
                            deltaLat) * 180 / pi;
      
      // 0-360도 정규화
      if (heading < 0) heading += 360;
      
      return heading;
    }
    return null;
  }
  
  // 즉시 보정: 꺼냈을 때 GPS 방향으로 리셋
  double instantCorrection(double sensorHeading) {
    if (gpsHeading != null && isMoving()) {
      // GPS 방향을 신뢰할 수 있는 경우
      if (gpsAccuracy < 10.0 && movementSpeed > 0.5) {
        // 센서를 GPS 방향으로 즉시 보정
        return gpsHeading!;
      }
    }
    return sensorHeading;
  }
}
```

#### 장점
- **자기장 간섭 없음**: GPS는 위성 신호 사용
- **누적 오차 없음**: 절대 위치 기반
- **즉시 보정 가능**: 이동 중이면 바로 방향 계산

### 2.2 하이브리드 즉시 보정 시스템

```dart
class InstantHeadingCorrection {
  // 상태 관리
  bool isInPocket = true;
  DateTime? deviceTakenOutTime;
  
  // GPS 트래킹 (주머니 모드에서도 동작)
  GPSHeadingCalibration gpsCalibration = GPSHeadingCalibration();
  
  // 나침반 빠른 안정화
  CompassStabilizer compass = CompassStabilizer();
  
  // 즉시 방향 획득 메인 함수
  double getInstantHeading() {
    // 1. 디바이스가 방금 꺼내졌는지 확인
    if (justTakenOut()) {
      return performInstantCalibration();
    }
    
    // 2. 이미 꺼낸 상태면 안정화된 나침반 사용
    return compass.getStabilizedHeading();
  }
  
  // 즉시 보정 수행
  double performInstantCalibration() {
    double? heading;
    
    // Priority 1: GPS 방향 (이동 중이면)
    if (isWalking() && gpsCalibration.hasValidGPSHeading()) {
      heading = gpsCalibration.gpsHeading;
      
      // 나침반과 자이로 리셋
      resetSensors(heading);
      return heading;
    }
    
    // Priority 2: 빠른 나침반 안정화 (정지 상태면)
    if (isStationary()) {
      // 8자 동작 없이 빠른 보정
      heading = compass.quickCalibration();
      return heading;
    }
    
    // Priority 3: 마지막 알려진 GPS 방향
    return gpsCalibration.lastKnownHeading ?? 0;
  }
  
  // 센서 리셋
  void resetSensors(double trueHeading) {
    // YawRate 리셋
    yawRate = trueHeading * (pi / 180);
    accumulatedDrift = 0;
    
    // 나침반 오프셋 설정
    compass.setOffset(trueHeading - compass.rawHeading);
  }
}
```

## 3. 🚶 주머니 모드 감지 및 관리

### 3.1 주머니 모드 자동 감지

```dart
class PocketModeDetector {
  // 센서 패턴으로 주머니 감지
  bool detectPocketMode() {
    // 1. 근접 센서 체크
    if (proximitySensor.isNear) {
      pocketProbability += 0.3;
    }
    
    // 2. 조도 센서 체크
    if (lightSensor.lux < 10) {
      pocketProbability += 0.3;
    }
    
    // 3. 가속도 패턴 (허벅지 스윙)
    if (detectThighSwingPattern()) {
      pocketProbability += 0.4;
    }
    
    return pocketProbability > 0.7;
  }
  
  // 허벅지 스윙 패턴 감지
  bool detectThighSwingPattern() {
    // 주머니에서 특징적인 가속도 패턴
    // Y축: 큰 진폭 (다리 스윙)
    // Z축: 작은 변화 (고정된 방향)
    double yVariance = calculateVariance(accelY);
    double zVariance = calculateVariance(accelZ);
    
    return yVariance > 2.0 && zVariance < 0.5;
  }
}
```

### 3.2 주머니 모드 중 GPS 트래킹

```dart
class PocketModeGPSTracking {
  // 주머니에서도 GPS는 계속 동작
  void continuousGPSTracking() {
    // 배터리 절약 모드
    if (isInPocket) {
      gps.updateInterval = 5000;  // 5초마다
      gps.desiredAccuracy = LocationAccuracy.balanced;
    } else {
      gps.updateInterval = 1000;  // 1초마다
      gps.desiredAccuracy = LocationAccuracy.high;
    }
    
    // GPS 이동 경로 기록
    gps.onLocationChanged.listen((position) {
      // 이동 방향 계산 및 저장
      if (lastPosition != null) {
        double distance = calculateDistance(lastPosition, position);
        
        if (distance > 2.0) {  // 2m 이상 이동
          gpsHeading = calculateBearing(lastPosition, position);
          lastValidGPSHeading = gpsHeading;
          lastHeadingTime = DateTime.now();
        }
      }
      lastPosition = position;
    });
  }
}
```

## 4. 📐 연구 기반 알고리즘

### 4.1 PCA (Principal Component Analysis) 방법
**논문**: "Heading Estimation for Indoor Pedestrian Navigation Using a Smartphone in the Pocket"

```dart
class PCAHeadingEstimation {
  // 주머니에서도 보행 방향 추정
  double estimateWalkingDirection() {
    // 1. 가속도 데이터 수집 (2초)
    List<Vector3> accelData = collectAcceleration(2000);
    
    // 2. Reference 좌표계로 투영
    List<Vector3> rcsData = projectToRCS(accelData);
    
    // 3. 수평면 가속도 추출
    List<Vector2> horizontalAccel = extractHorizontal(rcsData);
    
    // 4. PCA 적용
    Matrix2 covMatrix = calculateCovariance(horizontalAccel);
    Vector2 primaryAxis = getPrincipalComponent(covMatrix);
    
    // 5. 보행 방향 계산
    double walkingDirection = atan2(primaryAxis.y, primaryAxis.x);
    
    // 6. 180도 모호성 해결 (이전 GPS 방향 활용)
    return resolveAmbiguity(walkingDirection, lastGPSHeading);
  }
}
```

### 4.2 HDE (Heuristic Drift Elimination)
**특징**: 건물 내 복도가 직각으로 교차한다고 가정

```dart
class HeuristicDriftElimination {
  // 주요 방향 (0°, 90°, 180°, 270°)
  final List<double> dominantDirections = [0, 90, 180, 270];
  
  double applyHDE(double rawHeading) {
    // 가장 가까운 주요 방향으로 스냅
    double minDiff = 360;
    double snappedHeading = rawHeading;
    
    for (double dominant in dominantDirections) {
      double diff = (rawHeading - dominant).abs();
      if (diff < minDiff && diff < 15) {  // 15도 이내
        minDiff = diff;
        snappedHeading = dominant;
      }
    }
    
    return snappedHeading;
  }
}
```

## 5. 💡 구현 전략

### 5.1 SafeLight 통합 구현

```dart
// map_view_controller.dart 수정
class MapViewController extends BaseController {
  // 즉시 보정 시스템
  final InstantHeadingCorrection instantCorrection = InstantHeadingCorrection();
  final PocketModeDetector pocketDetector = PocketModeDetector();
  
  // GPS 트래킹
  Position? lastGPSPosition;
  double? gpsHeading;
  
  @override
  void onInit() {
    super.onInit();
    
    // GPS 연속 트래킹 시작
    startGPSTracking();
    
    // 주머니 모드 감지
    startPocketDetection();
  }
  
  // GPS 트래킹 (주머니에서도 동작)
  void startGPSTracking() {
    Geolocator.getPositionStream(
      locationSettings: LocationSettings(
        accuracy: LocationAccuracy.high,
        distanceFilter: 2,  // 2m 이상 이동시
      ),
    ).listen((position) {
      if (lastGPSPosition != null) {
        // GPS 방향 계산
        gpsHeading = Geolocator.bearingBetween(
          lastGPSPosition!.latitude,
          lastGPSPosition!.longitude,
          position.latitude,
          position.longitude,
        );
      }
      lastGPSPosition = position;
    });
  }
  
  // 즉시 방향 획득 (UI에서 호출)
  double getInstantHeading() {
    // 주머니에서 꺼냈는지 확인
    if (pocketDetector.justTakenOut()) {
      // GPS 방향으로 즉시 보정
      if (gpsHeading != null && isMoving()) {
        resetAllSensors(gpsHeading!);
        return gpsHeading!;
      }
    }
    
    // 일반 모드
    return fusedHeading;
  }
  
  // 모든 센서 리셋
  void resetAllSensors(double trueHeading) {
    // YawRate 리셋
    yawRate = trueHeading * angleToRadian;
    yawRate2 = yawRate;
    accumulatedDrift = 0;
    
    // Compass 오프셋
    double compassError = compassValue.value - trueHeading;
    compassOffset = compassError;
    
    // Fused heading 리셋
    fusedHeading = trueHeading;
    
    debugPrint('센서 리셋 완료: GPS 방향 = $trueHeading°');
  }
}
```

class BatteryOptimization {
  // 적응적 GPS 업데이트
  void adaptiveGPSUpdate() {
    if (isInPocket) {
      // 주머니: 저전력 모드
      gps.updateInterval = 5000;  // 5초
      gps.accuracy = LocationAccuracy.balanced;

    } else {
      // 활성: 정밀 모드
      gps.updateInterval = 1000;  // 1초
      gps.accuracy = LocationAccuracy.high;
      
      // IMU 센서 활성
      gyroscope.resume();
      accelerometer.samplingPeriod = 50;  // 50ms
    }
  }
}
